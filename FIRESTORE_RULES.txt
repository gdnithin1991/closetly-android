rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    function isSelf(userId) {
      return signedIn() && request.auth.uid == userId;
    }

    function isHouseholdMember(householdId) {
      return signedIn()
        && exists(/databases/$(database)/documents/households/$(householdId))
        && request.auth.uid in get(/databases/$(database)/documents/households/$(householdId)).data.memberIds;
    }

    function itemPhotoIsSmall() {
      return !("photoByteSize" in request.resource.data)
        || (
          request.resource.data.photoByteSize < 500 * 1024
          && request.resource.data.photoData.size() < 700 * 1024
        );
    }

    match /users/{userId} {
      allow read, write: if isSelf(userId);
    }

    match /householdInvites/{inviteCode} {
      allow create: if signedIn() && request.resource.data.createdBy == request.auth.uid;
      allow read: if signedIn();
      allow update: if signedIn();
    }

    match /households/{householdId} {
      allow create: if signedIn()
        && request.auth.uid in request.resource.data.memberIds
        && request.resource.data.createdBy == request.auth.uid;
      allow read, delete: if isHouseholdMember(householdId);
      allow update: if isHouseholdMember(householdId)
        || (
          signedIn()
          && request.resource.data.memberIds.hasAll(resource.data.memberIds)
          && request.auth.uid in request.resource.data.memberIds
        );

      match /items/{itemId} {
        allow read: if isHouseholdMember(householdId);
        allow write: if isHouseholdMember(householdId) && itemPhotoIsSmall();
      }

      match /sections/{sectionId} {
        allow read, write: if isHouseholdMember(householdId);
      }

      match /settings/{settingId} {
        allow read, write: if isHouseholdMember(householdId);
      }

      match /purchaseReminders/{reminderId} {
        allow read, write: if isHouseholdMember(householdId);
      }
    }
  }
}
